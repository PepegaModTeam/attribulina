# Pro Street PC gameplay Lua API. <kind> <name> <signature>  # description
# Types: number, string, boolean, table, instance (GRuntimeInstance userdata), simable (ISimable userdata),
# racerinfo (GRacerInfo userdata), handle (GWrapper userdata). See lua-api.md.
table Activity  # start and suspend activities
table Audio  # audio game-flow flags
table Bin  # career bin queries; stubs on PC
table Camera  # camera control
table Debug  # debug output; everything but ShowScreenMessage is a stub on PC
table Game  # most game functions: races, racers, triggers, cops, drift, UI flow
table HUD  # HUD messages
table Lua  # Lua runtime control
table LuaDebug  # debug console; a stub on PC
table Math  # RandomInt
table MiniMap  # minimap; a stub on PC
table Movie  # movie playback settings
table Music  # music control
table NIS  # in-game cinematics (NIS)
table Platform  # platform queries
table Racer  # per-racer control by index
table Render  # render effects
table Speech  # race speech
table Table  # Randomize
table Track  # track zone switching
table Career  # the career (GCareer userdata), set at startup; no methods are bound on PC
table GEvent  # the current race day event (GEvent userdata), set at startup
table GHub  # the current hub (GHub userdata), looked up on every read; nil outside a hub
table GMoviePlayer  # the movie player (GMoviePlayer userdata), set at startup
table GRaceParameters  # the current race's parameters (GRaceParameters userdata), looked up on every read; nil when there is none
table GTier  # the current tier (GTier userdata), looked up on every read; nil when there is none
table ProStreetGame  # the game (GProStreetGame userdata), set at startup
table RaceStatus  # the current race (GRaceStatus userdata), set at startup
